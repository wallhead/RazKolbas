cmake_minimum_required(VERSION 3.25)
project(RazKolbas VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
include(cmake/Dependencies.cmake)
add_library(rk_core STATIC src/plugin/Bootstrap.cpp)
target_include_directories(rk_core PUBLIC include)
if(MSVC)
  target_compile_options(rk_core PRIVATE /W4 /WX /permissive-)
endif()
if(WIN32)
  add_library(RazKolbas SHARED src/plugin/Plugin.cpp src/plugin/Logging.cpp)
  target_include_directories(RazKolbas SYSTEM PRIVATE "${commonlib_SOURCE_DIR}/include" "${rapidcsv_SOURCE_DIR}/src")
  target_compile_definitions(RazKolbas PRIVATE ENABLE_SKYRIM_SE=1 ENABLE_SKYRIM_AE=1 WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(RazKolbas PRIVATE rk_core spdlog::spdlog shell32 ole32)
  target_compile_options(RazKolbas PRIVATE /W4 /permissive- /wd4201 /Zi)
  target_link_options(RazKolbas PRIVATE /DEBUG)
endif()
if(BUILD_TESTING)
  add_executable(rk_tests tests/unit/HostTests.cpp)
  target_link_libraries(rk_tests PRIVATE rk_core Catch2::Catch2WithMain)
  add_test(NAME unit.host COMMAND rk_tests "[host]")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME integration.inventory COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${PROJECT_SOURCE_DIR}/tests/integration" -p test_inventory.py)
  if(WIN32)
    add_test(NAME integration.plugin COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/integration/test_plugin.py" "$<TARGET_FILE:RazKolbas>")
  endif()
endif()
